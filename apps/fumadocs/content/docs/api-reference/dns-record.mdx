---
title: DnsRecord
description: "DnsRecord type reference: record type, name, value, TTL, purpose (routing, ownership, certificate), and status, ready to show in a custom-domain settings UI."
---

```ts title="Type"
interface DnsRecord {
  type: "A" | "AAAA" | "CNAME" | "TXT" | "CAA" | "ALIAS" | "ANAME";
  name: string;
  value: string;
  ttl?: number;
  purpose: "routing" | "ownership" | "certificate" | "other";
  required: boolean;
  status: "pending" | "valid" | "invalid" | "unknown";
  description?: string;
}
```
